Section 03 · Java exam kit: reading input fast

On LeetCode you only write a method, and LeetCode passes you the input. The Infosys exam and Codeforces are different: you may have to write main yourself and read every number from standard input. If the Infosys platform gives you a get_ans(...) stub that already reads the input, keep its signature and just fill in the logic. If it doesn't, use FastScanner.

Why Scanner is slow

Scanner is convenient, but it checks every token against a regular expression and converts numbers in a locale-aware way, and it reads the input through a small 1024-character buffer. FastScanner uses BufferedReader, which reads big chunks and returns a whole line at once, and StringTokenizer, which just splits that line on spaces. FastReader goes further: it reads raw bytes 64 KB at a time and builds each number digit by digit, without creating any Strings.

I timed all three on this machine (OpenJDK 25), reading numbers written one per line like the Infosys input. Each figure is the middle of three runs, reading time only:

Reader10⁵ numbers10⁶ numbersWhen to use it
Scanner182 ms818 msFine for small inputs. Risky above about 10⁵ numbers when the time limit is 1–2 s.
FastScanner (BufferedReader + StringTokenizer)46 ms144 msMemorise this one. About 5× faster than Scanner, short, and handles strings and whole lines.
FastReader (byte buffer)10 ms33 msOptional. Only needed when you read millions of numbers.

Output matters too. Printing 10⁶ lines with System.out.println took 2.0 s; the same output through one PrintWriter took 0.15 s.

FastScanner: the version to memorise

📄Main.javatype from memory
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        FastScanner fs = new FastScanner(System.in);
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));

        int n = fs.nextInt();                    // works for "one value per line" and "all on one line"
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = fs.nextLong();

        long sum = 0;
        for (long x : a) sum += x;
        out.println(sum);
        out.flush();                             // without this you print nothing
    }

    /** BufferedReader reads big chunks; StringTokenizer splits a line on spaces. */
    static final class FastScanner {
        private final BufferedReader br;
        private StringTokenizer st;

        FastScanner(InputStream in) {
            br = new BufferedReader(new InputStreamReader(in));
        }

        String next() throws IOException {
            while (st == null || !st.hasMoreTokens()) {   // current line used up: read another
                String line = br.readLine();
                if (line == null) return null;           // end of input
                st = new StringTokenizer(line);
            }
            return st.nextToken();
        }

        int nextInt() throws IOException { return Integer.parseInt(next()); }

        long nextLong() throws IOException { return Long.parseLong(next()); }

        double nextDouble() throws IOException { return Double.parseDouble(next()); }

        String nextLine() throws IOException {           // the next whole line, spaces included
            st = null;                                   // (drops anything left on the current line)
            return br.readLine();
        }
    }
}
  1. new BufferedReader(new InputStreamReader(System.in)) reads standard input in large blocks instead of a few characters at a time.
  2. next() returns one word. If the current line has no words left, it reads the next line and wraps it in a StringTokenizer. That's why it doesn't matter whether the numbers come one per line or all on one line.
  3. nextInt(), nextLong(), nextDouble() take that word and convert it.
  4. nextLine() returns the next whole line, spaces included. Use it only for text such as a sentence; it throws away anything left on the current line.
  5. PrintWriter collects your output in memory. out.flush() at the end sends it all at once. Forget it and nothing is printed.

FastReader (byte version, optional) 10× faster than FastScanner; harder to type from memory
import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    public static void main(String[] args) throws IOException {
        FastReader in = new FastReader(System.in);
        PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));

        int n = in.nextInt();              // works whether values are on one line or one per line
        long[] a = new long[n];
        for (int i = 0; i < n; i++) a[i] = in.nextLong();

        long sum = 0;
        for (long x : a) sum += x;
        out.println(sum);

        out.flush();                        // forgetting this = empty output = 0 marks
    }

    /** Token reader: skips spaces AND newlines, so "one value per line" input is handled. */
    static final class FastReader {
        private final DataInputStream in;
        private final byte[] buf = new byte[1 << 16];
        private int len = 0, ptr = 0;

        FastReader(InputStream is) { in = new DataInputStream(is); }

        private int read() throws IOException {
            if (ptr == len) {
                len = in.read(buf, 0, buf.length);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buf[ptr++];
        }

        long nextLong() throws IOException {
            int c = read();
            while (c != '-' && (c < '0' || c > '9')) {
                if (c == -1) throw new EOFException();
                c = read();
            }
            boolean neg = c == '-';
            if (neg) c = read();
            long r = 0;
            while (c >= '0' && c <= '9') { r = r * 10 + (c - '0'); c = read(); }
            return neg ? -r : r;
        }

        int nextInt() throws IOException { return (int) nextLong(); }

        String next() throws IOException {
            int c = read();
            while (c != -1 && c <= ' ') c = read();
            if (c == -1) throw new EOFException();
            StringBuilder sb = new StringBuilder();
            while (c > ' ') { sb.append((char) c); c = read(); }
            return sb.toString();
        }
    }
}

read() refills a 64 KB byte buffer whenever it runs out. nextLong() skips everything that isn't a digit or a minus sign, then builds the number one digit at a time (r = r * 10 + digit). It never creates a String, which is where its speed comes from.

Deep recursion on a big-stack thread (fallback)
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Run the real work on a thread with a 128 MB stack so deep recursion doesn't crash.
        Thread t = new Thread(null, () -> {
            try { solve(); } catch (IOException e) { throw new UncheckedIOException(e); }
        }, "solver", 1 << 27);
        t.start();
    }

    static void solve() throws IOException {
        int n = 1_000_000;                                  // a path of 1e6 nodes
        System.out.println(depth(n));
    }

    static int depth(int k) { return k == 0 ? 0 : 1 + depth(k - 1); }
}

Tested with recursion depth 10⁶. Use it only when rewriting without recursion would take too long.

Five full mocks

Each mock is 3 hours in one sitting under the same rules. The Codeforces problems were picked by rating to match the Q1/Q2/Q3 steps, and none of them appears in the daily lists. Codeforces has no partial marks, but its verdict tells you the first failing test ("Wrong answer on test 7"); use that to estimate the share of tests you'd pass.

Mock rules
  1. Timer at 180 minutes, phone in another room.
  2. Minutes 0–15: read every problem. On paper, write the limits, your idea and its complexity for each.
  3. Solve the problem you're surest of first. That isn't always Q1.
  4. Test on the samples with Codeforces Custom test before you submit.
  5. By minute 150, every problem has a submission, even if it's only a brute force.
After the mock (about 60 minutes)
  1. Enter your scores below (estimated % of tests passed).
  2. For each unsolved problem, read the editorial, close it, and write the solution yourself.
  3. Log the cause of each failure: misread, wrong idea, overflow, off-by-one, TLE or input.
  4. Pick the weakest topic and redo 2 of its core problems the next day.

RPG: maximum monsters defeatedVery high (11 reports) · SP · 2022–2026

You start with e experience. Monster i needs experience ≥ power[i] and adds bonus[i] when defeated. You can fight them in any order, and losing a fight ends the run. Find the most monsters you can defeat. Input order: n, e, then n powers, then n bonuses, one value per line.

💡 Hint
If you can beat a set of monsters, you can beat them in order of increasing power.
✅ Approach
Sort the (power, bonus) pairs by power. Walk through them while e ≥ power, adding each bonus. Use long for e. O(n log n). This works because bonuses are non-negative; check the limits. Practice twin: LC 502 IPO.
Count arrays where each element divides the nextVery high (13) · DSE · 2021–2025

Count arrays of length K with values in 1..N where a[i] divides a[i+1]. Print the count mod 10000. Samples: (N=2, K=1) → 2, (2, 2) → 3, (3, 2) → 5.

💡 Hint
Small K and N: dp[v] pushed to every multiple of v, in O(K·N log N). Large K: an array has at most ⌊log₂N⌋+1 distinct values.
✅ Approach
Count strictly increasing divisor chains with s distinct values ending at each v, using DP over multiples. Each chain extends to a length-K array in C(K−1, s−1) ways, depending on where the values change. 10000 isn't prime, so build C with Pascal's triangle, not inverses. The same problem with mod 10⁹+7 is LC 2338.
Longest non-decreasing subsequence with XOR ≥ MVery high (13) · DSE, SP · 2022–2026

This is the Hard problem of the official sample paper. The full solution is in the sample-paper walkthroughs; open them only after Mock 4.

Maximum XOR choosing at most N/2 elementsVery high (13) · SP · 2021–2026

N is even (N ≤ 120) and A[i] ≤ 10⁶. Choose at most N/2 elements, not necessarily consecutive, to maximise their XOR. Samples: [1, 2] → 2 (not 3, because of the cap); [1, 2, 4, 7] → 7.

💡 Hint
A ≤ 10⁶ < 2²⁰, so there are only about a million possible XOR values. What's the fewest number of elements that produces each one?
✅ Approach
XOR knapsack: dp[x] = fewest elements whose XOR is x. For each element a, new[x] = min(old[x], old[x ^ a] + 1). That's O(N·2²⁰) ≈ 1.3·10⁸ simple steps. The answer is the largest x with dp[x] ≤ N/2. A shortcut: a linear basis has at most 20 numbers, so when N ≥ 40 the cap never binds and the XOR basis answer is correct.
Smallest base where M has all identical digitsVery high (10) · SP · 2021–2026

M ≤ 10¹². Find the smallest base B in which every digit of M is the same. Samples: 41 → 40 ("11"), 34430 → 312 ("110 110").

💡 Hint
For B > 10⁶, B² > 10¹² ≥ M, so M has at most two digits in that base.
✅ Approach
Check bases 2..10⁶ directly; each check costs O(log M). If none works, M must be "dd" in base B, so M = d·(B+1). For each divisor d ≤ 10⁶ of M, B = M/d − 1 is valid when B > d. Take the smallest such B. B = M − 1 ("11") always works for M ≥ 3. The all-ones version is LC 483.
Pack gifts into exactly K consecutive boxes, maximise the total of distinct countsVery high (9) · SP · 2022–2026

N ≤ 35000, K ≤ 50. This is CF 833B The Bakery with the same limits (Day 21 bonus). The handbook card Split into K parts has a tested solution. The O(K·N²) DP gets the small tests.

Split into k segments, minimise the sum of (last − first occurrence)PrepInsta SP set

n ≤ 35000, k ≤ 100. This is CF 1527E Partition Game (Day 21 bonus). When a[i] repeats, it adds i − prev[a[i]] to every split point before prev[a[i]], which is a range add.

Andy's vacation: cancel at most K obligationsHigh (5) · SP · 2021–2025

Days 1..N (N ≤ 10⁶) and M obligations on given days; several can fall on the same day. Cancel at most K of them and find the longest run of free consecutive days. Samples: (N=10, K=2, days 6 9 3 2 7) → 5; (N=7, K=0, days 3 4) → 3.

✅ Approach
Count obligations per day, then slide a window over the days while the window's count is ≤ K. O(N + M). Pattern twin: LC 1004. The trap: two obligations on one day must both be cancelled to free that day.
Minimum work to transport wine between housesHigh (6) · SP · 2022–2026

Houses in a line each buy (+) or sell (−) bottles, and the total is zero. Moving one bottle one house costs 1. Find the minimum total work. Judge: GfG Wine Buying and Selling.

✅ Approach
The answer is the sum of |prefix sum| over every gap between neighbouring houses, because each gap carries exactly the imbalance to its left. Use long.
Minimum railway platformsVery high (8) · SP · 2022–2026

Given arrival and departure times, find the fewest platforms so that no train waits. This is the maximum overlap, the same as LC 2406 (Day 4). Check whether a train arriving at the exact minute another departs needs a new platform; the statement decides.

Arrange numbers to form the largest numberHigh (4) · SP · 2023–2026

This is LC 179 (Day 4). Sort strings with the comparator (b + a).compareTo(a + b) and return "0" if every number is zero.

Choose X in [0, K] to maximise Σ (X xor A[i])High (6) · 2021–2025
💡 Hint
Bits are independent in a sum of XORs. Setting bit b of X to 1 contributes (n − cnt₁[b])·2ᵇ; setting it to 0 contributes cnt₁[b]·2ᵇ.
✅ Approach
Walk K's bits from the top, as in digit DP. While you match K exactly, each 1-bit of K gives one candidate: put 0 there and choose freely below, taking the better contribution for each lower bit. Also count X = K itself. Take the best candidate. O(31·31) after counting bits. Use long.
Gym: fewest exercises to drain energy E (each exercise usable at most twice)High (5) · SP · 2020–2025
✅ Approach
Sort the drains in descending order and use each one twice until E ≤ 0. If E is still positive after all of them, print −1. The greedy is correct because only the count matters, so the largest drains are always best.
Longest increasing subsequence where every adjacent pair has (x & y)·2 < (x | y)High (4) · SP

Samples: [15, 6, 5, 12, 1] → 2; [9, 17, 2, 15, 5, 2] → 2; [17, 16, 12, 2, 8, 17, 17] → 3.

💡 Hint
Try the condition on pairs whose highest set bit is the same, and on pairs where it differs.
✅ Approach
The condition holds exactly when x and y have different highest set bits (I checked every pair up to 600). So the highest bit must strictly increase along the subsequence, and its length is at most 31. Keep best[b] = longest chain ending with highest bit b; each element costs 1 + max(best[0..h−1]). O(31·n).
Fewest days to make terrain strictly decreasing (day t allows changes of 2^(t−1))High (6) · SP · 2021–2025

The core idea is the same as CF 1338A Powered Addition: find the largest deficit you must fix; the answer is that deficit's bit length. Adjust for "strictly" and for the direction of the change.

Largest subset of [1, N−1] whose product ≡ 1 (mod N)High (4) · 2023–2025

This is CF 1514C. Take every number coprime to N. If their product mod N isn't 1, drop the element equal to that product.

EASY · Tree beauty: perfect-square pairs per subtree
  1. x·y is a perfect square exactly when x and y have the same square-free kernel (the product of the primes with odd exponent). So the task becomes: count pairs with equal kernels in every subtree.
  2. Kernels for values up to 10⁹: divide out the primes ≤ 1000 (only 168 of them). What remains has no prime factor ≤ 1000 and is below 1009³, so it's 1, q, q² or q·r. q² drops out; anything else stays in the kernel.
  3. Small-to-large: each node keeps a map kernel → count. Merge the smaller map into the larger one. Adding c copies of a key the big map already holds b times creates b·c new pairs. Total O(n log n).
  4. Process nodes in reverse BFS order. A recursive DFS overflows the stack on a 10⁵-node chain.
import java.io.*;
import java.util.*;

/** Infosys sample paper, EASY: sum over u of (# pairs in subtree(u) whose product is a perfect square). */
public class Main {
    static final long MOD = 1_000_000_007L;
    static int[] smallPrimes;

    public static void main(String[] args) throws IOException {
        FastReader in = new FastReader(System.in);
        int n = in.nextInt();
        int[] par = new int[n + 1];
        for (int v = 1; v <= n; v++) par[v] = in.nextInt();
        long[] a = new long[n + 1];
        for (int v = 1; v <= n; v++) a[v] = in.nextLong();
        System.out.println(solve(n, par, a));
    }

    static long solve(int n, int[] par, long[] a) {
        buildSmallPrimes();
        // children lists + BFS order (no recursion: a 1e5 chain would overflow the stack)
        int[] head = new int[n + 1], nxt = new int[n + 1];
        Arrays.fill(head, -1);
        int root = 1;
        for (int v = 1; v <= n; v++) {
            if (par[v] == 0) root = v;
            else { nxt[v] = head[par[v]]; head[par[v]] = v; }
        }
        int[] order = new int[n];
        int qh = 0, qt = 0;
        order[qt++] = root;
        while (qh < qt) { int u = order[qh++]; for (int c = head[u]; c != -1; c = nxt[c]) order[qt++] = c; }

        @SuppressWarnings("unchecked")
        HashMap<Long, Integer>[] mp = new HashMap[n + 1];
        long[] pairs = new long[n + 1];
        for (int v = 1; v <= n; v++) { mp[v] = new HashMap<>(); mp[v].put(kernel(a[v]), 1); }

        long ans = 0;
        for (int i = n - 1; i >= 0; i--) {          // children are finished before their parent
            int v = order[i];
            ans += pairs[v];                          // beauty(v) is final now
            int p = par[v];
            if (p == 0) continue;
            HashMap<Long, Integer> big = mp[p], small = mp[v];
            if (big.size() < small.size()) { HashMap<Long, Integer> t = big; big = small; small = t; }
            long cross = 0;
            for (Map.Entry<Long, Integer> e : small.entrySet()) {
                int have = big.getOrDefault(e.getKey(), 0);
                cross += (long) have * e.getValue();
                big.put(e.getKey(), have + e.getValue());
            }
            pairs[p] += pairs[v] + cross;
            mp[p] = big;
            mp[v] = null;
        }
        return ans % MOD;
    }

    /** Square-free kernel: x*y is a perfect square  <=>  kernel(x) == kernel(y). */
    static long kernel(long x) {
        long k = 1;
        for (int p : smallPrimes) {                  // primes <= 1000 only (168 of them)
            if ((long) p * p > x) break;
            int e = 0;
            while (x % p == 0) { x /= p; e ^= 1; }
            if (e == 1) k *= p;
        }
        // x has no prime factor <= 1000 left (or x < p*p, so x is 1 or prime)
        // x <= 1e9 < 1009^3, so x is 1, q, q*q or q*r for primes q, r > 1000
        long s = (long) Math.sqrt((double) x);
        while (s * s > x) s--;
        while ((s + 1) * (s + 1) <= x) s++;
        if (s * s != x) k *= x;                      // q or q*r stays in the kernel; q*q vanishes
        return k;
    }

    static void buildSmallPrimes() {
        if (smallPrimes != null) return;
        List<Integer> ps = new ArrayList<>();
        for (int i = 2; i <= 1000; i++) {
            boolean pr = true;
            for (int d = 2; d * d <= i; d++) if (i % d == 0) { pr = false; break; }
            if (pr) ps.add(i);
        }
        smallPrimes = ps.stream().mapToInt(Integer::intValue).toArray();
    }

    static final class FastReader {
        private final DataInputStream in; private final byte[] buf = new byte[1 << 16]; private int len, ptr;
        FastReader(InputStream is) { in = new DataInputStream(is); }
        private int read() throws IOException {
            if (ptr == len) { len = in.read(buf, 0, buf.length); ptr = 0; if (len <= 0) return -1; }
            return buf[ptr++];
        }
        long nextLong() throws IOException {
            int c = read();
            while (c != '-' && (c < '0' || c > '9')) { if (c == -1) throw new EOFException(); c = read(); }
            boolean neg = c == '-'; if (neg) c = read();
            long r = 0; while (c >= '0' && c <= '9') { r = r * 10 + (c - '0'); c = read(); }
            return neg ? -r : r;
        }
        int nextInt() throws IOException { return (int) nextLong(); }
    }
}

Partial-credit fallback: for each node, collect its subtree and test all pairs. That's O(n³) and passes only tiny tests, but it beats a blank submission.

MEDIUM · Good subsequence with gcd exactly p, under updates
  1. Only multiples of p can be in the subsequence. Let g be the gcd of all multiples of p. If g ≠ p, every subset has gcd ≥ g > p, so the answer is NO.
  2. If g = p and at least one element isn't a multiple of p, take all the multiples. That subsequence is shorter than n, so the answer is YES.
  3. If every element is a multiple of p: divide by p, so the gcd of everything is 1. A value ≤ 10⁵ has at most 6 distinct primes (2·3·5·7·11·13·17 > 10⁵), so some subset of at most 7 elements already has gcd 1. With n ≥ 8 the answer is YES; with n ≤ 7, try removing each element.
  4. Keep g in a gcd segment tree in which non-multiples are stored as 0 (gcd(0, x) = x), plus a counter of non-multiples. Each query costs O(log n).
  5. Input order: n, p, the array, q, the column count (2), then the queries. This differs from the written format (Lesson 2).
import java.io.*;
import java.util.*;

/**
 * Infosys sample paper, MEDIUM: after each point update, does a non-empty subsequence
 * of length < n with gcd exactly p exist? Count YES answers.
 * NOTE: the samples put p BEFORE the array (the PDF's written format says after it).
 */
public class Main {
    public static void main(String[] args) throws IOException {
        FastReader in = new FastReader(System.in);
        int n = in.nextInt();
        int p = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        int q = in.nextInt();
        int cols = in.nextInt();                    // always 2, read and ignore
        int[][] qs = new int[q][2];
        for (int i = 0; i < q; i++) { qs[i][0] = in.nextInt(); qs[i][1] = in.nextInt(); }
        System.out.println(solve(n, p, a, qs));
    }

    static long solve(int n, int p, int[] a, int[][] qs) {
        a = a.clone();
        int size = 1; while (size < n) size <<= 1;
        int[] t = new int[2 * size];               // gcd of elements divisible by p (others stored as 0)
        int notDiv = 0;
        for (int i = 0; i < n; i++) {
            if (a[i] % p == 0) t[size + i] = a[i]; else notDiv++;
        }
        for (int i = size - 1; i > 0; i--) t[i] = gcd(t[2 * i], t[2 * i + 1]);

        long yes = 0;
        for (int[] qu : qs) {
            int i = qu[0] - 1, v = qu[1];
            if (a[i] % p != 0) notDiv--;
            a[i] = v;
            if (a[i] % p != 0) notDiv++;
            int pos = size + i;
            t[pos] = (a[i] % p == 0) ? a[i] : 0;
            for (pos >>= 1; pos > 0; pos >>= 1) t[pos] = gcd(t[2 * pos], t[2 * pos + 1]);

            int g = t[1];                            // gcd of all multiples of p (0 if none)
            boolean ok;
            if (g != p) ok = false;                  // every subset gcd is a multiple of g
            else if (notDiv > 0) ok = true;          // take all multiples: length < n
            else if (n >= 8) ok = true;              // a/p <= 1e5 has <= 6 primes -> gcd-1 subset of size <= 7
            else ok = existsDropOne(a, p);           // n <= 7: try removing each element
            if (ok) yes++;
        }
        return yes;
    }

    static boolean existsDropOne(int[] a, int p) {
        int n = a.length;
        for (int skip = 0; skip < n; skip++) {
            int g = 0;
            for (int j = 0; j < n; j++) if (j != skip) g = gcd(g, a[j]);
            if (g == p) return true;
        }
        return false;
    }

    static int gcd(int x, int y) { while (y != 0) { int r = x % y; x = y; y = r; } return x; }

    static final class FastReader {
        private final DataInputStream in; private final byte[] buf = new byte[1 << 16]; private int len, ptr;
        FastReader(InputStream is) { in = new DataInputStream(is); }
        private int read() throws IOException {
            if (ptr == len) { len = in.read(buf, 0, buf.length); ptr = 0; if (len <= 0) return -1; }
            return buf[ptr++];
        }
        long nextLong() throws IOException {
            int c = read();
            while (c != '-' && (c < '0' || c > '9')) { if (c == -1) throw new EOFException(); c = read(); }
            boolean neg = c == '-'; if (neg) c = read();
            long r = 0; while (c >= '0' && c <= '9') { r = r * 10 + (c - '0'); c = read(); }
            return neg ? -r : r;
        }
        int nextInt() throws IOException { return (int) nextLong(); }
    }
}
HARD · Longest non-decreasing subsequence with XOR ≥ M
  1. Values are ≤ N ≤ 1000 < 1024, so any XOR fits in 10 bits. Add that XOR as a DP dimension.
  2. Keep fw[x], a Fenwick tree over values storing the longest subsequence with XOR x that ends in a value ≤ v.
  3. For each element v and each target x, look up the best length with XOR x ^ v ending at a value ≤ v, then add 1. Compute all 1024 candidates before writing any of them, so v isn't used twice.
  4. Cost: N · 1024 · log N ≈ 2·10⁷ steps. The answer is the maximum over x ≥ M.
import java.io.*;
import java.util.*;

/** Infosys sample paper, HARD: longest non-decreasing subsequence whose XOR is >= M. */
public class Main {
    public static void main(String[] args) throws IOException {
        FastReader in = new FastReader(System.in);
        int n = in.nextInt();
        int m = in.nextInt();
        int[] a = new int[n];
        for (int i = 0; i < n; i++) a[i] = in.nextInt();
        System.out.println(solve(a, m));
    }

    // fw[x] = Fenwick over values; prefix max of "longest subsequence with XOR x ending in value <= v"
    static int solve(int[] a, int m) {
        int maxV = 1;
        for (int x : a) maxV = Math.max(maxV, x);
        int B = 1;
        while (B <= maxV) B <<= 1;                   // every XOR of values <= maxV is < B
        int[][] fw = new int[B][maxV + 1];
        int[] cand = new int[B];
        for (int v : a) {
            for (int x = 0; x < B; x++) {
                int best = query(fw[x ^ v], v);      // extend a subsequence whose XOR was x ^ v
                cand[x] = best > 0 ? best + 1 : 0;
            }
            cand[v] = Math.max(cand[v], 1);          // start a new subsequence with v alone
            for (int x = 0; x < B; x++) if (cand[x] > 0) update(fw[x], v, cand[x]);
        }
        int ans = 0;
        for (int x = m; x < B; x++) ans = Math.max(ans, query(fw[x], maxV));
        return ans;
    }

    static int query(int[] f, int i) { int r = 0; for (; i > 0; i -= i & -i) r = Math.max(r, f[i]); return r; }
    static void update(int[] f, int i, int v) { for (; i < f.length; i += i & -i) if (f[i] < v) f[i] = v; }

    static final class FastReader {
        private final DataInputStream in; private final byte[] buf = new byte[1 << 16]; private int len, ptr;
        FastReader(InputStream is) { in = new DataInputStream(is); }
        private int read() throws IOException {
            if (ptr == len) { len = in.read(buf, 0, buf.length); ptr = 0; if (len <= 0) return -1; }
            return buf[ptr++];
        }
        long nextLong() throws IOException {
            int c = read();
            while (c != '-' && (c < '0' || c > '9')) { if (c == -1) throw new EOFException(); c = read(); }
            boolean neg = c == '-'; if (neg) c = read();
            long r = 0; while (c >= '0' && c <= '9') { r = r * 10 + (c - '0'); c = read(); }
            return neg ? -r : r;
        }
        int nextInt() throws IOException { return (int) nextLong(); }
    }
}
